# The data a Tricot stores: the schema of a new data file, and what the migrations make of each release's database.
# Written once and never regenerated: a later change to what the core stores must leave this file as it is.
# test/dataGolden.ts says how it is made, and test/dataGolden.test.ts compares it.

## a new data file
data version 21
index block_versions_by_asset on block_versions: CREATE INDEX block_versions_by_asset ON block_versions (asset) WHERE asset IS NOT NULL
index events_by_actor on events: CREATE INDEX events_by_actor ON events (board, actor_kind, actor_name, seq)
index events_by_decision on events: CREATE INDEX events_by_decision ON events (board, decision_id, seq) WHERE decision_id IS NOT NULL
index events_by_item on events: CREATE INDEX events_by_item ON events (board, item_id, seq) WHERE item_id IS NOT NULL
index events_by_task on events: CREATE INDEX events_by_task ON events (board, task_id, seq) WHERE task_id IS NOT NULL
index sqlite_autoindex_boards_1 on boards: (automatic)
index sqlite_autoindex_tricot_skills_1 on tricot_skills: (automatic)
index sqlite_autoindex_tricot_skills_2 on tricot_skills: (automatic)
table agent_skills on agent_skills: CREATE TABLE agent_skills (
          agent       TEXT PRIMARY KEY,
          name        TEXT NOT NULL,
          skills      TEXT NOT NULL,
          reported_at TEXT NOT NULL
        ) STRICT, WITHOUT ROWID
table block_versions on block_versions: CREATE TABLE block_versions (
          board TEXT NOT NULL,
          block TEXT NOT NULL,
          rev   INTEGER NOT NULL,
          at    TEXT NOT NULL,
          body  TEXT NOT NULL,
          asset TEXT GENERATED ALWAYS AS (CASE WHEN json_extract(body, '$.type') = 'image' THEN json_extract(body, '$.assetId') END) VIRTUAL,
          PRIMARY KEY (board, block, rev)
        ) STRICT, WITHOUT ROWID
table boards on boards: CREATE TABLE boards (
          key  TEXT PRIMARY KEY,
          id   TEXT NOT NULL,
          pos  INTEGER NOT NULL,
          body TEXT NOT NULL
        ) STRICT
table data_versions on data_versions: CREATE TABLE data_versions (
          part    TEXT PRIMARY KEY,
          version INTEGER NOT NULL
        ) STRICT, WITHOUT ROWID
table events on events: CREATE TABLE events (
          board       TEXT NOT NULL,
          seq         INTEGER NOT NULL,
          at          TEXT NOT NULL,
          body        TEXT NOT NULL,
          type        TEXT GENERATED ALWAYS AS (json_extract(body, '$.type')) VIRTUAL,
          actor_kind  TEXT GENERATED ALWAYS AS (json_extract(body, '$.actor.kind')) VIRTUAL,
          actor_name  TEXT GENERATED ALWAYS AS (json_extract(body, '$.actor.name')) VIRTUAL,
          decision_id TEXT GENERATED ALWAYS AS (json_extract(body, '$.decisionId')) VIRTUAL,
          task_id     TEXT GENERATED ALWAYS AS (json_extract(body, '$.taskId')) VIRTUAL,
          item_id     TEXT GENERATED ALWAYS AS (json_extract(body, '$.itemId')) VIRTUAL,
          summary     TEXT GENERATED ALWAYS AS (json_extract(body, '$.summary')) VIRTUAL,
          PRIMARY KEY (board, seq)
        ) STRICT, WITHOUT ROWID
table items on items: CREATE TABLE items (
          board TEXT NOT NULL,
          owner TEXT NOT NULL,
          kind  TEXT NOT NULL,
          id    TEXT NOT NULL,
          pos   INTEGER NOT NULL,
          body  TEXT NOT NULL,
          PRIMARY KEY (board, owner, kind, id)
        ) STRICT, WITHOUT ROWID
table need_dismissals on need_dismissals: CREATE TABLE need_dismissals (
          person TEXT NOT NULL,
          board  TEXT NOT NULL,
          key    TEXT NOT NULL,
          at     TEXT NOT NULL,
          PRIMARY KEY (person, board, key)
        ) STRICT, WITHOUT ROWID
table tricot_skills on tricot_skills: CREATE TABLE tricot_skills (
          id           TEXT PRIMARY KEY,
          key          TEXT NOT NULL UNIQUE,
          name         TEXT NOT NULL,
          description  TEXT NOT NULL,
          instructions TEXT NOT NULL,
          agent        TEXT,
          created_by   TEXT NOT NULL,
          created_at   TEXT NOT NULL,
          updated_at   TEXT NOT NULL
        ) STRICT
